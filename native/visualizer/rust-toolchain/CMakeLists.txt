cmake_minimum_required(VERSION 3.19)
project(PlezyMilkRustToolchain LANGUAGES C CXX)
if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSROOT OR NOT IS_DIRECTORY "${CMAKE_SYSROOT}")
  message(FATAL_ERROR "MilkRust TV builds require an explicit SDK cross compiler and sysroot")
endif()
set(tools "{}")
foreach(name CMAKE_C_COMPILER CMAKE_CXX_COMPILER CMAKE_AR CMAKE_SYSROOT CMAKE_C_FLAGS CMAKE_CXX_FLAGS CMAKE_EXE_LINKER_FLAGS)
  set(value "${${name}}")
  string(REPLACE "\\" "\\\\" value "${value}")
  string(REPLACE "\"" "\\\"" value "${value}")
  string(REPLACE "\n" "\\n" value "${value}")
  string(JSON tools SET "${tools}" "${name}" "\"${value}\"")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/rust-tools.json" "${tools}\n")
