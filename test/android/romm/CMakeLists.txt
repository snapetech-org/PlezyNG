cmake_minimum_required(VERSION 3.22.1)
project(romm_android_tests LANGUAGES C CXX)
set(PLEZY_ROMM_BUILD_CORES OFF CACHE BOOL "" FORCE)
set(PLEZY_ROMM_ISOLATED ON CACHE BOOL "" FORCE)
add_subdirectory(../../../native/games/romm romm)
target_compile_definitions(plezy_retro PRIVATE PLEZY_RETRO_TEST_NO_AUDIO)
include(FetchContent)
FetchContent_GetProperties(romm_libretro_common SOURCE_DIR common)
# The fixture stands in for one bundled core; it is never packaged in the app.
add_library(gambatte_libretro SHARED ../../native/romm_libretro_fixture.cpp)
target_include_directories(gambatte_libretro PRIVATE "${common}/include")
target_compile_features(gambatte_libretro PRIVATE cxx_std_17)
add_library(romm_android_test SHARED romm_android_test.cpp)
target_link_libraries(romm_android_test PRIVATE plezy_retro)
target_compile_features(romm_android_test PRIVATE cxx_std_17)
