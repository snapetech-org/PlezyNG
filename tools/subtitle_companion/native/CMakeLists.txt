cmake_minimum_required(VERSION 3.20)
project(plezy_subtitle_decode C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET
  "libavformat>=60" "libavformat<63" "libavcodec>=60" "libavcodec<63" "libavutil>=58" "libavutil<61")
add_executable(plezy-subtitle-decode bitmap_decode.c)
target_link_libraries(plezy-subtitle-decode PRIVATE PkgConfig::FFMPEG)
if(MSVC)
  target_compile_options(plezy-subtitle-decode PRIVATE /W4)
else()
  target_compile_options(plezy-subtitle-decode PRIVATE -Wall -Wextra -Wconversion -Werror=implicit-function-declaration)
endif()
install(TARGETS plezy-subtitle-decode RUNTIME DESTINATION bin)
